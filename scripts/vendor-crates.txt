# vendor/ in the workspace Cargo.toml must be listed (scripts/gate.sh --check).

# are not independently buildable.

# crate <dir> <level>
#   strict: cargo fmt --check, clippy --all-targets -D warnings, cargo test

#           clippy error, listed in the gate summary but not failing in the
#           untouched upstream files (scripts/vendor-lint.sh)

#   a file of a `warn` crate changed by a DeltaForge patch, held to the
#   strict rules; every `src/...rs` named in the crate's DELTAFORGE-PATCH.md
#   must be listed
# extra <dir> <name> <command...>
#   another command run in the copy (its own target directory), for example
#   a loom model.
